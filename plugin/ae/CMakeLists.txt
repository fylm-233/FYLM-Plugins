 COMMAND "${CMAKE_COMMAND}" "-DCOMPILER=${CMAKE_CXX_COMPILER}" "-DSDK=${AE_EXAMPLES}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/DitheringStudioPiPL.r" "-DOUTPUT=${PIPL_RC}" "-DMFR=${DITHER_ENABLE_MFR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/pipl.cmake"

# 0.4 product split. The old combined target is retained for regression/old-project recovery,
# and is deliberately excluded from the FYLM distribution.
math(EXPR FYLM_PACKED_VERSION "(${PROJECT_VERSION_MAJOR} << 19) | (${PROJECT_VERSION_MINOR} << 15) | (${PROJECT_VERSION_PATCH} << 11) | 1")
configure_file(FYLMVersion.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/FYLMVersion.hpp" @ONLY)
foreach(product IN ITEMS DitheringStudio CRTVHS ShaderGlass)
 set(target "FYLM_${product}")
 if(product STREQUAL "ShaderGlass")
  set(FYLM_NAME "FYLM ShaderGlass")
  set(FYLM_MATCH "FYLM_ShaderGlass")
  set(FYLM_FLAGS "0x06000004")
 elseif(product STREQUAL "CRTVHS")
  set(FYLM_NAME "FYLM CRT VHS")
  set(FYLM_MATCH "FYLM_CRTVHS")
  set(FYLM_FLAGS "0x06000004")
 else()
  set(FYLM_NAME "FYLM Pixel Art")
  set(FYLM_MATCH "FYLM_DitheringStudio")
  set(FYLM_FLAGS "0x06000000")
 endif()
 set(resource "${CMAKE_CURRENT_BINARY_DIR}/${target}.r")
 set(FYLM_FLAGS2 "0x00000480")
 set(FYLM_FLAGS2_MFR "0x08000480")
 if(product STREQUAL "DitheringStudio")
  set(FYLM_FLAGS2 "0x00001480")
  set(FYLM_FLAGS2_MFR "0x08001480")
 endif()
 set(rc "${CMAKE_CURRENT_BINARY_DIR}/${target}.rc")
 configure_file(FYLMPiPL.r.in "${resource}" @ONLY)
 add_custom_command(OUTPUT "${rc}"
  COMMAND "${CMAKE_COMMAND}" "-DCOMPILER=${CMAKE_CXX_COMPILER}" "-DSDK=${AE_EXAMPLES}" "-DSOURCE=${resource}" "-DOUTPUT=${rc}" "-DMFR=${DITHER_ENABLE_MFR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/pipl.cmake"
  DEPENDS "${resource}" pipl.cmake VERBATIM)
 add_library(${target} MODULE FYLMEffects.cpp "${rc}")
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 target_include_directories(${target} PRIVATE ../../modules/retro_gpu/include "${AE_EXAMPLES}/Headers" "${AE_EXAMPLES}/Headers/SP" "${AE_EXAMPLES}/Util")
 target_compile_definitions(${target} PRIVATE MSWindows WIN32 NOMINMAX)
 target_link_libraries(${target} PRIVATE dither_core dither_i18n)
 if(product STREQUAL "DitheringStudio")
  target_link_libraries(${target} PRIVATE pixel_art_core pixel_art_presets comdlg32)
  target_include_directories(${target} PRIVATE "${CMAKE_SOURCE_DIR}/third_party/ShaderGlass-master/ShaderGC/include")
 endif()
 if(product STREQUAL "CRTVHS")
  target_link_libraries(${target} PRIVATE fylm_crt_runtime)
  target_compile_definitions(${target} PRIVATE FYLM_CRT_ONLY=1)
 endif()
 if(product STREQUAL "ShaderGlass")
  add_dependencies(${target} fylm_shaderglass_presets)
  target_link_libraries(${target} PRIVATE fylm_shaderglass comdlg32)
  target_compile_definitions(${target} PRIVATE FYLM_SHADERGLASS=1)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_SOURCE_DIR}/assets/ShaderGlass" "$<TARGET_FILE_DIR:${target}>/Assets/ShaderGlass")
 endif()
 set_target_properties(${target} PROPERTIES PREFIX "" SUFFIX ".aex")
 if(product STREQUAL "DitheringStudio")
  # Renamed binary, frozen identity: two files registering FYLM_DitheringStudio must never
  # coexist in the AE scan path, so the installer removes the old name before copying this one.
  set_target_properties(${target} PROPERTIES OUTPUT_NAME "FYLM_PixelArt")
 endif()
 add_executable(${target}_ui_schema ../../tools/emit_fylm_ui.cpp)
 if(product STREQUAL "CRTVHS")
  target_compile_definitions(${target}_ui_schema PRIVATE FYLM_CRT_ONLY=1)
 endif()
 if(product STREQUAL "ShaderGlass")
  target_compile_definitions(${target}_ui_schema PRIVATE FYLM_SHADERGLASS=1)
 endif()
 set(schema "${CMAKE_BINARY_DIR}/panel/${target}.jsxinc")
 add_custom_command(OUTPUT "${schema}" COMMAND ${target}_ui_schema "${schema}" DEPENDS ${target}_ui_schema FYLMParams.hpp VERBATIM)
 add_custom_target(${target}_panel ALL DEPENDS "${schema}")
 if(DITHER_ENABLE_MFR)
  target_compile_definitions(${target} PRIVATE DITHER_ENABLE_MFR=1)
 endif()
 if(BUILD_TESTING)
  add_executable(${target}_tests ../../tests/test_fylm_adapter.cpp)
  target_include_directories(${target}_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_include_directories(${target}_tests PRIVATE ../../modules/retro_gpu/include "${AE_EXAMPLES}/Headers" "${AE_EXAMPLES}/Headers/SP" "${AE_EXAMPLES}/Util")
  target_compile_definitions(${target}_tests PRIVATE MSWindows WIN32 NOMINMAX)
  target_link_libraries(${target}_tests PRIVATE dither_core dither_i18n)
  if(product STREQUAL "DitheringStudio")
   target_link_libraries(${target}_tests PRIVATE pixel_art_core pixel_art_presets comdlg32)
   target_include_directories(${target}_tests PRIVATE "${CMAKE_SOURCE_DIR}/third_party/ShaderGlass-master/ShaderGC/include")
  endif()
  if(product STREQUAL "CRTVHS")
   target_link_libraries(${target}_tests PRIVATE fylm_crt_runtime)
   target_compile_definitions(${target}_tests PRIVATE FYLM_CRT_ONLY=1)
  endif()
  if(product STREQUAL "ShaderGlass")
   target_link_libraries(${target}_tests PRIVATE fylm_shaderglass comdlg32)
    target_compile_definitions(${target}_tests PRIVATE FYLM_SHADERGLASS=1 SG_TEST_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  endif()
  if(product STREQUAL "ShaderGlass")
   add_test(NAME shaderglass_no_external_assets COMMAND "${CMAKE_COMMAND}" "-DPLUGIN=$<TARGET_FILE:${target}>" "-DTEST=$<TARGET_FILE:${target}_tests>" "-DWORK=${CMAKE_BINARY_DIR}/sg-no-assets" -P "${CMAKE_SOURCE_DIR}/tests/shaderglass_embedded.cmake")
  endif()
  add_dependencies(${target}_tests ${target})
  add_test(NAME ${target}_adapter COMMAND ${target}_tests "$<TARGET_FILE:${target}>")
 endif()
endforeach()
configure_file("panel/FYLM Plugins.jsx" "${CMAKE_BINARY_DIR}/panel/FYLM Plugins.jsx" COPYONLY)
configure_file("${CMAKE_BINARY_DIR}/generated/i18n-catalog.jsxinc" "${CMAKE_BINARY_DIR}/panel/FYLM-i18n.jsxinc" COPYONLY)

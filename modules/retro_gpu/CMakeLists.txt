cmake_minimum_required(VERSION 3.21)
project(RetroIntegrationKit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(BUILD_TESTING)
 set(retro_checks_default ON)
elseif(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
 set(retro_checks_default ON)
else()
 set(retro_checks_default OFF)
endif()
option(RETRO_BUILD_CHECKS "Build standalone module checks" ${retro_checks_default})
if(RETRO_BUILD_CHECKS)
 enable_testing()
endif()
foreach(pair IN ITEMS "crt;crt_vhs.glsl" "dither;dither.glsl" "vertex;fullscreen.vert.glsl" "entry;unified.frag.glsl")
 list(GET pair 0 name)
 list(GET pair 1 file)
 file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${file}" ${name})
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${file}")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
configure_file(src/embedded_shaders.in.hpp "${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_shaders.hpp" @ONLY)
add_library(retro_gpu_module STATIC src/retro.cpp)
target_include_directories(retro_gpu_module PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties(retro_gpu_module PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(retro_gpu_module PUBLIC /utf-8 PRIVATE /W4)
endif()
if(WIN32)
 target_sources(retro_gpu_module PRIVATE src/renderer_win.cpp)
 target_compile_definitions(retro_gpu_module PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(retro_gpu_module PUBLIC opengl32)
endif()
if(RETRO_BUILD_CHECKS)
add_executable(retro_module_tests tests/test_module.cpp)
target_link_libraries(retro_module_tests PRIVATE retro_gpu_module)
add_test(NAME retro_controls_plan_uniforms COMMAND retro_module_tests)
add_executable(retro_emit_shaders tools/emit_shaders.cpp)
target_link_libraries(retro_emit_shaders PRIVATE retro_gpu_module)
set(shader_outputs "${CMAKE_CURRENT_BINARY_DIR}/shaders/fullscreen.vert")
foreach(index RANGE 0 6)
 list(APPEND shader_outputs "${CMAKE_CURRENT_BINARY_DIR}/shaders/pass${index}.frag")
endforeach()
add_custom_command(OUTPUT ${shader_outputs}
 COMMAND retro_emit_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders"
 DEPENDS retro_emit_shaders VERBATIM)
add_custom_target(retro_shader_sources ALL DEPENDS ${shader_outputs})
find_program(GLSLC NAMES glslc)
if(GLSLC)
 foreach(index RANGE 0 6)
  add_test(NAME glsl_pass_${index} COMMAND "${GLSLC}" --target-env=opengl -fauto-bind-uniforms -fauto-map-locations -fshader-stage=fragment "${CMAKE_CURRENT_BINARY_DIR}/shaders/pass${index}.frag" -o "${CMAKE_CURRENT_BINARY_DIR}/shaders/pass${index}.spv")
 endforeach()
 add_test(NAME glsl_vertex COMMAND "${GLSLC}" --target-env=opengl -fauto-map-locations -fshader-stage=vertex "${CMAKE_CURRENT_BINARY_DIR}/shaders/fullscreen.vert" -o "${CMAKE_CURRENT_BINARY_DIR}/shaders/vertex.spv")
else()
 message(STATUS "GLSLC not found: GLSL compilation tests unavailable (C++ tests still build)")
endif()
set(AE_SDK_ROOT "" CACHE PATH "Optional Adobe SDK root for compiling parameter examples")
if(AE_SDK_ROOT)
 add_library(retro_ae_compile_check OBJECT ../../plugin/ae/retro/params_example.cpp)
 target_include_directories(retro_ae_compile_check PRIVATE include "${AE_SDK_ROOT}/Examples/Headers" "${AE_SDK_ROOT}/Examples/Headers/SP" "${AE_SDK_ROOT}/Examples/Util")
 target_compile_definitions(retro_ae_compile_check PRIVATE MSWindows WIN32 NOMINMAX)
 target_link_libraries(retro_ae_compile_check PRIVATE retro_gpu_module)
endif()

endif() # RETRO_BUILD_CHECKS

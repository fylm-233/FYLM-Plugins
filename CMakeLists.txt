cmake_minimum_required(VERSION 3.21)
project(DitheringStudio VERSION 0.14.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dither_core STATIC core/src/render.cpp core/src/palettes.cpp core/src/halftone.cpp)
target_include_directories(dither_core PUBLIC core/include)
set_target_properties(dither_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(dither_core PRIVATE /W4 /fp:strict)
else()
 target_compile_options(dither_core PRIVATE -Wall -Wextra -ffp-contract=off)
endif()
# PA-M1 pixel-art core: independent precision and colour contract, no host dependencies.
add_library(pixel_art_core STATIC core/src/pixel_art/render.cpp core/src/pixel_art/engine.cpp core/src/pixel_art/structure.cpp core/src/pixel_art/parallel.cpp core/src/pixel_art/execution.cpp core/src/pixel_art/pc98_color.cpp)
find_package(Threads REQUIRED)
target_link_libraries(pixel_art_core PUBLIC Threads::Threads)
target_include_directories(pixel_art_core PUBLIC core/include)
set_target_properties(pixel_art_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(pixel_art_core PRIVATE /W4 /fp:strict)
else()
 target_compile_options(pixel_art_core PRIVATE -Wall -Wextra -ffp-contract=off)
endif()
include(i18n/catalog.cmake)

# PA-014 CLI half of the host/CLI integration; the AE path joins in PA-M2 with the first
# exposed style, so no unreachable UI is shipped in the meantime.

target_link_libraries(pixel_art_cli PRIVATE pixel_art_core dither_core)
